# B05: compile-to-JS languages (ReScript, PureScript, Pyret). All three come from npm (pinned);
# PureScript's three base packages come from their GitHub release tags (fetched once here, at build time).
RUN chronic apt-get -qq install -y npm && chronic apt-get clean
# Only the two compilers are linked into PATH: the prefix .bin also has "rc" and "which", which must not shadow the real ones.
RUN npm install --prefix /opt/b05 --no-fund --no-audit rescript@12.3.1 purescript@0.15.17 pyret-npm@0.1.29 && for b in bsc purs; do ln -s /opt/b05/node_modules/.bin/$b /usr/local/bin/$b; done
RUN for p in prelude:v6.0.2 effect:v4.0.0 console:v6.1.0; do mkdir -p /opt/purs/lib/${p%%:*} && curl -fsSL https://github.com/purescript/purescript-${p%%:*}/archive/refs/tags/${p#*:}.tar.gz | tar -xz --strip-components=1 -C /opt/purs/lib/${p%%:*} || exit 1; done
RUN printf '#!/bin/sh\n# pyretc IN.arr OUT.jarr: compile one Pyret program to a standalone node file\n# (the same compiler call the pyret CLI makes, without its background compile server)\nI=$(realpath "$1"); O=$(realpath -m "$2")\ncd %s && exec node --max-old-space-size=8192 --stack-size=8192 build/phaseA/pyret.jarr --build-runnable "$I" --outfile "$O" --builtin-js-dir src/js/trove/ --builtin-arr-dir src/arr/trove --require-config src/scripts/standalone-configA.json --standalone-file src/js/base/handalone.js --compiled-dir /opt/pyret-compiled -no-check-mode < /dev/null\n' /opt/b05/node_modules/pyret-npm/pyret-lang > /usr/local/bin/pyretc && chmod +x /usr/local/bin/pyretc && printf 'print("ok")' > /tmp/w.arr && pyretc /tmp/w.arr /tmp/w.jarr && NODE_PATH=/opt/b05/node_modules node /tmp/w.jarr && rm -f /tmp/w.arr /tmp/w.jarr && chmod -R a+rwX /opt/pyret-compiled
