# B13: embeddable scripting I (Janet, Berry, Gravity, Umka, Wren, Io). All small pinned source builds.
RUN cd /opt && git clone -q --depth 1 -b v1.43.0 https://github.com/janet-lang/janet && make -C janet -j4 > /tmp/janet.log 2>&1 && rm -f /tmp/janet.log && ln -s /opt/janet/build/janet /usr/local/bin/janet && echo '(prin "janet ok")' > /tmp/t.janet && janet /tmp/t.janet && rm -f /tmp/t.janet
RUN cd /opt && git clone -q --depth 1 -b v1.1.0 https://github.com/berry-lang/berry && make -C berry -j4 > /tmp/berry.log 2>&1 && rm -f /tmp/berry.log && ln -s /opt/berry/berry /usr/local/bin/berry && echo 'print("berry ok")' > /tmp/t.be && berry /tmp/t.be && rm -f /tmp/t.be
RUN cd /opt && git clone -q --depth 1 -b 0.9.9 https://github.com/marcobambini/gravity && make -C gravity -j4 > /tmp/gravity.log 2>&1 && rm -f /tmp/gravity.log && ln -s /opt/gravity/gravity /usr/local/bin/gravity && echo 'func main(){System.print("gravity ok")}' > /tmp/t.gravity && gravity -q /tmp/t.gravity && rm -f /tmp/t.gravity
RUN cd /opt && git clone -q --depth 1 -b v1.5.7 https://github.com/vtereshkov/umka-lang && make -C umka-lang -j4 > /tmp/umka.log 2>&1 && rm -f /tmp/umka.log && ln -s /opt/umka-lang/build/umka /usr/local/bin/umka && echo 'fn main(){printf("umka ok")}' > /tmp/t.um && umka /tmp/t.um && rm -f /tmp/t.um
# wren-cli 0.4.0: its vm.c defines a static write() that clashes with unistd.h on new glibc/gcc; rename it (2 places).
RUN cd /opt && git clone -q --depth 1 -b 0.4.0 https://github.com/wren-lang/wren-cli && sed -i 's/static void write(/static void cliWrite(/; s/= write;/= cliWrite;/' wren-cli/src/cli/vm.c && make -C wren-cli/projects/make -j4 > /tmp/wren.log 2>&1 && rm -f /tmp/wren.log && ln -s /opt/wren-cli/bin/wren_cli /usr/local/bin/wren_cli && echo 'System.print("wren ok")' > /tmp/t.wren && wren_cli /tmp/t.wren && rm -f /tmp/t.wren
# Io tag 2019.05.22-alpha (last classic CMake release; glibc dropped sys/sysctl.h).
RUN cd /opt && git clone -q --depth 1 -b 2019.05.22-alpha --recurse-submodules --shallow-submodules https://github.com/IoLanguage/io && sed -i 's|^# include <sys/sysctl.h>|/* sys/sysctl.h dropped */|' io/libs/iovm/source/IoSystem.c && mkdir io/build && cd io/build && (cmake .. -DCMAKE_BUILD_TYPE=Release -DCMAKE_POLICY_VERSION_MINIMUM=3.5 '-DCMAKE_C_FLAGS=-std=gnu11 -Wno-error=implicit-function-declaration -Wno-error=incompatible-pointer-types -Wno-error=int-conversion' > /tmp/io.log 2>&1 && make -j4 io_static >> /tmp/io.log 2>&1 || (grep -n -i -m20 -B2 'error' /tmp/io.log; tail -15 /tmp/io.log; exit 1)) && cp _build/binaries/io_static /usr/local/bin/io && rm -f /tmp/io.log && echo '"io ok" println' > /tmp/t.io && io /tmp/t.io && rm -f /tmp/t.io
